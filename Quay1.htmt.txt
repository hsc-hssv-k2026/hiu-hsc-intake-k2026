<script>
const API_URL = "https://script.google.com/macros/s/AKfycbyAkoch5dGeF12mUJpC5swPGaGCfkwUpywant4OYSBQwJGySPFfANWToV67AHvbi2_D/exec";
const API_KEY = "HIU-K2026-QuayTiepNhan-7Xk2m9";
var currentQueue = [];

function render() {
  var filter = document.getElementById('filter').value.trim().toLowerCase();
  var list = document.getElementById('list');
  var filtered = (currentQueue || []).filter(function (s) {
    if (!filter) return true;
    return (s.mssv + ' ' + s.hoten).toLowerCase().indexOf(filter) !== -1;
  });
  document.getElementById('count').textContent = currentQueue.length;
  document.getElementById('empty').style.display = filtered.length ? 'none' : 'block';
  list.innerHTML = '';
  filtered.forEach(function (s) {
    var div = document.createElement('div');
    div.className = 'row';
    div.innerHTML =
      '<div class="info"><b>' + s.hoten + '</b><br>MSSV: ' + s.mssv +
      ' &middot; ' + (s.nganh || '') + '</div>';
    var btn = document.createElement('button');
    btn.className = 'act';
    btn.textContent = 'Xác nhận - mời qua Quầy 2';
    btn.onclick = function () { confirmStudent(s.mssv, btn); };
    div.appendChild(btn);
    list.appendChild(div);
  });
}

function showMsg(text, ok) {
  var msg = document.getElementById('msg');
  msg.style.display = 'block';
  msg.className = ok ? 'ok' : 'err';
  msg.textContent = text;
  setTimeout(function () { msg.style.display = 'none'; }, 4000);
}

function confirmStudent(mssv, btn) {
  btn.disabled = true;
  btn.textContent = 'Đang xử lý...';
  var nhanvien = document.getElementById('nhanvien').value.trim();

  fetch(API_URL + "?action=quay1_confirm", {
    method: "POST",
    headers: { "Content-Type": "text/plain;charset=utf-8" },
    body: JSON.stringify({ mssv: mssv, nhanvien: nhanvien, key: API_KEY })
  })
    .then(function (response) { return response.json(); })
    .then(function (res) {
      showMsg(res.message, res.success);
      loadQueue();
    })
    .catch(function (err) {
      showMsg('Lỗi hệ thống: ' + err.message, false);
      btn.disabled = false;
      btn.textContent = 'Xác nhận - mời qua Quầy 2';
    });
}

function loadQueue() {
  fetch(API_URL + "?action=layDanhSachQuay1&key=" + encodeURIComponent(API_KEY))
    .then(function (response) { return response.json(); })
    .then(function (res) {
      currentQueue = res || [];
      render();
    })
    .catch(function (err) {
      showMsg('Lỗi tải danh sách: ' + err.message, false);
    });
}

document.getElementById('filter').addEventListener('input', render);

loadQueue();
setInterval(loadQueue, 5000); // tự làm mới mỗi 5 giây
</script>
